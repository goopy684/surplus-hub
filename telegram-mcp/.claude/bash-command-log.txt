import asyncio, json, os
from dotenv import load_dotenv
from pathlib import Path
from telethon import TelegramClient

load_dotenv()
API_ID = int(os.environ['TELEGRAM_API_ID'])
API_HASH = os.environ['TELEGRAM_API_HASH']
SESSION_PATH = str(Path('.') / 'telegram_session')

async def main():
    client = TelegramClient(SESSION_PATH, API_ID, API_HASH)
    await client.start()
    
    entity = await client.get_entity(-1002470324395)
    
    from datetime import datetime, timezone, timedelta
    since = datetime.now(timezone.utc) - timedelta(days=7)
    
    messages = await client.get_messages(entity, limit=500, offset_date=None)
    filtered = [m for m in messages if m.date and m.date >= since]
    
    for msg in filtered:
        sender_name = ''
        if msg.sender:
            if hasattr(msg.sender, 'first_name'):
                parts = [msg.sender.first_name or '', msg.sender.last_name or '']
                sender_name = ' '.join(p for p in parts if p)
            elif hasattr(msg.sender, 'title'):
                sender_name = msg.sender.title
        date_str = msg.date.strftime('%m/%d %H:%M')
        text = (msg.text or '').replace(chr(10), ' | ')
        reply = f' [reply:{msg.reply_to_msg_id}]' if msg.reply_to else ''
        print(f'[{date_str}] {sender_name}: {text}{reply}')
    
    print(f'\\n--- Total: {len(filtered)} messages in last 7 days ---')
    await client.disconnect()

asyncio.run(main())

from pathlib import Path
from telethon import TelegramClient
from datetime import datetime, timezone, timedelta

load_dotenv()

API_HASH = os.environ['TELEGRAM_API_HASH']
SESSION_PATH = str(Path('.') / 'telegram_session')

async def main():
    client = TelegramClient(SESSION_PATH, API_ID, API_HASH)
    await client.start()
    
    entity = await client.get_entity(-1003839782776)
    since = datetime.now(timezone.utc) - timedelta(days=7)
    
    messages = await client.get_messages(entity, limit=500)
    filtered = [m for m in messages if m.date and m.date >= since]
    
    for msg in filtered:
        sender_name = ''
        if msg.sender:
            if hasattr(msg.sender, 'first_name'):
                parts = [msg.sender.first_name or '', msg.sender.last_name or '']
                sender_name = ' '.join(p for p in parts if p)
            elif hasattr(msg.sender, 'title'):
                sender_name = msg.sender.title
        date_str = msg.date.strftime('%m/%d %H:%M')
        text = (msg.text or '').replace(chr(10), ' | ')

        print(f'[{date_str}] {sender_name}: {text}{reply}')

    await client.disconnect()
